-- giải path dạng chuỗi trong DB thành Instance (tự lành khi game đổi tên chi tiết nhỏ)
local function resolve(scope, dotted, wantProp)
	local segs = {}
	for part in tostring(dotted):gmatch("[^%.]+") do segs[#segs + 1] = part end
	local cur = scope
	for i = 1, #segs do
		if not cur then return nil end
		local part = segs[i]
		local nxt
		if part == "PlayerGui" then nxt = LPR:FindFirstChild("PlayerGui")
		elseif part == "ReplicatedStorage" then nxt = RS
		elseif part == "workspace" or part == "Workspace" then nxt = workspace
		elseif part == "LocalPlayer" then nxt = LPR
		elseif cur.FindFirstChild then
			nxt = cur:FindFirstChild(part)
			-- path trong dump có thể kết thúc bằng TÊN THUỘC TÍNH (vd "...QuestTitle.Title.Text")
			if not nxt and i == #segs then
				local okp, val = pcall(function() return cur[part] end)
				if okp and val ~= nil then
					return { __prop = part, inst = cur }
				end
			end
		end
		cur = nxt
	end
	return cur
end

local guiPath = ((BF_OFFSETS.live or {}).gui or {})
local pathCache = {}
local function guiNode(key, ttl)
	local e = pathCache[key]
	if e and tick() - e.at < (ttl or 1.5) then
		if e.ok then return e.inst, true end
		return nil, false
	end
	local def = guiPath[key]
	local inst = def and resolve(LPR, def) or nil
	pathCache[key] = { at = tick(), inst = inst, ok = inst ~= nil }
	return inst, inst ~= nil
end
-- đọc giá trị: nếu node là "thuộc tính cuối path" thì đọc đúng thuộc tính đó
local function guiValue(key, fallback)
	local n = guiNode(key)
	if not n then return fallback end
	if type(n) == "table" and n.__prop then
		local ok, v = pcall(function() return n.inst[n.__prop] end)
		if ok and v ~= nil then return v end
		return fallback
	end
	return n
end
local function guiText(key)
	local n = guiNode(key)
	if not n then return nil end
	if type(n) == "table" and n.__prop then
		local ok, v = pcall(function() return n.inst[n.__prop] end)
		if ok and v ~= nil then return tostring(v) end
		return nil
	end
	local ok, v = pcall(function() return n.Text end)
	if ok and v ~= nil and tostring(v) ~= "" then return tostring(v) end
	local ok2, v2 = pcall(function() return n.ContentText end)
	return ok2 and v2 or nil
end

local COMMF_LAST_OK = nil
local function commF(name, ...)
	local c = getCommF()
	if not c then return nil end
	local args = { ... }
	local ok, res = pcall(function() return c:InvokeServer(name, UNPACK(args)) end)
	COMMF_LAST_OK = ok and (res ~= nil) and name or nil
	if ok then return res end
	return nil
end
local function remoteAt(p, method, ...)
	local r = resolve(game, p)
	if not r then return nil end
	local args = { ... }
	local ok, res = pcall(function()
		if r.ClassName:find("RemoteFunction") then return r:InvokeServer(UNPACK(args)) end
		r:FireServer(UNPACK(args)); return true
	end)
	if ok then return res end
	return nil
end

local function useSkill(k)
	if k == "C" then
		safe(function() VIM:SendKeyEvent(true, Enum.KeyCode.C, false, game) end)
		safe(function() VIM:SendKeyEvent(false, Enum.KeyCode.C, false, game) end)
		return
	end
	local c = LPR.Character
	if not c then return end
	for _, t in ipairs(c:GetChildren()) do
		if t:IsA("Tool") then
			local rf = t:FindFirstChild("RemoteFunction")
			if rf then
				if k == "Z" then
					local r = root()
					local cf = r and CFrame.new(r.Position, mousePos()) or CFrame.new()
					safe(function() rf:InvokeServer("Z", cf) end)
				else
					safe(function() rf:InvokeServer(k) end)
				end
			end
		end
	end
end

local function teleportTo(target, entrance)
	local r = root()
	if not r then return false end
	local p
	if typeof(target) == "Vector3" then p = target
	elseif typeof(target) == "CFrame" then p = target.Position
	elseif typeof(target) == "Instance" then
		if target:IsA("BasePart") then p = target.Position
		elseif target:IsA("Model") then p = target:GetPivot().Position end
	end
	if not p then return false end
	if entrance then
		local e = V(entrance)
		if e then remoteAt(BF_OFFSETS.live.paths.commF, "requestEntrance", e) task.wait(0.35) end
	end
	local hv = hum()
	if hv then hv.WalkSpeed = 0 end
	local dest = CFrame.new(p + Vector3.new(0, 6, 0))
	for _ = 1, 3 do
		safe(function() r.CFrame = dest r.AssemblyLinearVelocity = Vector3.zero end)
		task.wait(0.07)
	end
	task.wait(0.1)
	if hv and not CFG.Noclip then hv.WalkSpeed = CFG.WalkSpeed end
	return true
end

	local l = name:match("%[Lv%.?%s*(%d+)%]")
	return l and tonumber(l) or nil
end

-- Duump thật: workspace.Enemies chứa cả mob KHÔNG có [Lv.] (Trainee) → phải nhận hết
local cacheE, cacheEAt = {}, 0
local function scanEnemies(force)
	if not force and tick() - cacheEAt < 0.7 and #cacheE > 0 then return cacheE end
	cacheEAt = tick()
	local out = {}
	local function take(folder)
		for _, m in ipairs(folder:GetChildren()) do
			if m:IsA("Model") and not isPlayerChar(m) then
				local h = m:FindFirstChildOfClass("Humanoid")
				local r = m:FindFirstChild("HumanoidRootPart")
				if h and r and h.Health > 0 then
					out[#out + 1] = { model = m, hrp = r, hum = h, name = m.Name, base = mobBase(m.Name),
						lv = mobLv(m.Name), hp = h.MaxHealth or 0, folder = folder.Name }
				end
			end
		end
	end
	local f = workspace:FindFirstChild("Enemies")
	if f then take(f) end
	local b = workspace:FindFirstChild("Bosses")
	if b then take(b) end
	if #out == 0 then
		for _, m in ipairs(workspace:GetChildren()) do
			if m:IsA("Model") then
				local h = m:FindFirstChildOfClass("Humanoid")
				if h and m:FindFirstChild("HumanoidRootPart") and not isPlayerChar(m) then
					out[#out + 1] = { model = m, hrp = m:FindFirstChild("HumanoidRootPart"), hum = h,
						name = m.Name, base = mobBase(m.Name), lv = mobLv(m.Name), hp = h.MaxHealth or 0, folder = "workspace" }
				end
			end
		end
	end
	cacheE = out
	return out
end
local BOSS_HINTS = { "boss", "king", "lord", "admiral", "warden", "cyborg", "swan", "trevor",
	"empress", "greybeard", "diamond", "order", "mad", "iron", "pirate", "captain", "queen", "reaper" }
local function isBoss(e)
	local n = e.base:lower()
	for _, k in ipairs(BOSS_HINTS) do if n:find(k, 1, true) then return true end end
	if e.hp >= 5000 then return true end
	local sea = onSea()
	for _, b in ipairs(BF_OFFSETS.bosses[sea] or {}) do
		if n:find(b.name:lower(), 1, true) or b.name:lower():find(n, 1, true) then return true end
	end
	return false
end
local function findMobByName(name)
	local want = tostring(name):lower()
	local all = scanEnemies()
	for _, e in ipairs(all) do if e.base:lower() == want or e.name:lower() == want then return e end end
	for _, e in ipairs(all) do
		if e.base:lower():find(want, 1, true) or want:find(e.base:lower(), 1, true) then return e end
	end
	return nil
end
local function nearestEnemy(pred, maxD)
	maxD = maxD or 1e9
	local best, bd
	for _, e in ipairs(scanEnemies()) do
		if not pred or pred(e) then
			local d = (e.hrp.Position - pos()).Magnitude
			if d < maxD and (not bd or d < bd) then best, bd = e, d end
		end
	end
	return best
end

-- FRUIT: dump thật cho thấy phải LỌC mesh rác ("Bloxfruits_SkyBonus6_Cube")
local FR_EX = { "bloxfruit", "meshes", "/", "fruitdealer", "fruitshop", "fruitview", "fruitnotifier", "fruitinventory" }

	if not inst or not inst.Parent then return false end
	local n = inst.Name:lower()

	if inst:IsA("Tool") then return true end
	if inst:IsA("Model") or inst:IsA("BasePart") then
		if not inst.Name:find("%a+%-%a+") then return false end
		if inst:IsA("Model") and not inst:FindFirstChild("Handle") and not inst:FindFirstChildWhichIsA("BasePart") then return false end
		return true
	end
	return false
end
local function partPos(inst)
	if not inst then return nil end
	if inst:IsA("BasePart") then return inst.Position end
	local h = inst:FindFirstChild("Handle") or inst:FindFirstChildWhichIsA("BasePart")
	if h then return h.Position end
	local hrp = inst:FindFirstChild("HumanoidRootPart")
	if hrp then return hrp.Position end
	local ok, pv = pcall(function() return inst:GetPivot().Position end)

end

local function questVisible()
	if guiPath.questVisible then
		local v = guiValue("questVisible")
		if type(v) == "boolean" then return v end
	end
	local q = guiNode("quest")
	if not q then return false end
	local ok, v = pcall(function() return q.Visible end)
	return ok and v == true
end
local function questTitle()
	local t = guiText("questTitle")
	if type(t) ~= "string" then return nil end
	return t
end
local function singular(w)
	if not w then return nil end
	local s = w:gsub("%s+$", "")
	if #s > 3 and s:sub(-1) == "s" and s:sub(-2) ~= "ss" then s = s:sub(1, -2) end
	return s
end
-- trả về: target(mob), số cần giết, đã giết, tổng, tên quest gốc
local function questInfo()
	local t = questTitle()
	if not t then return nil end
	local need, mob, cur, max = t:match("DEFEAT%s+(%d+)%s+([%a' ]-)%s*%((%d+)%s*/%s*(%d+)%)")
	if not need then
		need, mob, cur, max = t:match("Defeat%s+(%d+)%s+([%a' ]-)%s*%((%d+)%s*/%s*(%d+)%)")
	end
	if not need then
		mob = t:match("[Qq]uest:%s*(.-)%s*%[Lv") or t:match("[Dd]efeat%s+%d+%s+([%a' ]+)")
		if mob then return { title = t, mob = singular(mob:gsub("%s+$", "")), cur = 0, max = 1, need = 0 } end
		return { title = t, mob = nil }
	end
	return { title = t, mob = singular(mob), need = tonumber(need), cur = tonumber(cur) or 0,
		max = tonumber(max) or tonumber(need) or 0 }
end

local function questCandidates(lv)
	local sea = onSea()
	local list = (BF_OFFSETS.quests[sea] or {})
	local out = {}
	for _, q in ipairs(list) do
		if q.id and (q.lv or 0) > 0 and math.abs((q.lv or 0) - lv) <= 60 then
			out[#out + 1] = q
		end
	end
	table.sort(out, function(a, b) return math.abs(a.lv - lv) < math.abs(b.lv - lv) end)
	return out
end

local function questGiverNear()
	local best, bd
	local function try(m)
		if m:IsA("Model") and not isPlayerChar(m) then
			local hrp = child(m, "HumanoidRootPart")
			if hrp then
				local isG = m:FindFirstChild("QuestGiver") ~= nil or m:FindFirstChild("TitleQuest") ~= nil
					or m:FindFirstChildWhichIsA("ProximityPrompt", true) ~= nil
					or m:FindFirstChild("ClickDetector") ~= nil
				if isG then
					local d = dist(hrp.Position)
					if not bd or d < bd then best, bd = m, d end
				end
			end
		end
	end
	for _, fname in ipairs({ "NPCs", "Enemies", "Characters", "Quest" }) do
		local f = workspace:FindFirstChild(fname)
		if f then for _, m in ipairs(f:GetChildren()) do try(m) end end
	end
	return best
end

-- Cờ "đang bận": nếu vì lý do gì đó không được reset (lỗi giữa đường) thì tự hết hạn sau 15s
local function busyOK()

		CFG._busy = false
	end
	return not CFG._busy
end
local function setBusy(v)
	CFG._busy = v and true or false
	CFG._busyAt = tick()
end

local function acceptQuestBody()
	local ok = false
	local lv = lvl()

	-- (1) server tự chỉ NPC quest tốt nhất
	local hint = remoteAt(BF_OFFSETS.live.paths.getBestQuestNPC, nil)
	local targetPos
	if type(hint) == "table" then
		local hrp = child(hint, "HumanoidRootPart")
		local p = hint.Position or hint.pos or (hrp and hrp.Position)
		if typeof(p) == "Vector3" then targetPos = p end
		if not targetPos and hint.Name then
			local m = workspace:FindFirstChild(hint.Name, true)
			local hrp = m and child(m, "HumanoidRootPart")
			if hrp then targetPos = hrp.Position end
		end
	elseif type(hint) == "string" then
		local m = workspace:FindFirstChild(hint, true)
		local hrp = m and child(m, "HumanoidRootPart")
		if hrp then targetPos = hrp.Position end
	end

	-- (2) NPC quest gần mình
	if not targetPos then
		local g = questGiverNear()
		local hrp = g and child(g, "HumanoidRootPart")
		if hrp then targetPos = hrp.Position end
	end

	-- (3) DB: toạ độ NPC theo level
	if not targetPos then
		local cands = questCandidates(lv)
		if cands[1] and V(cands[1].npc) then targetPos = V(cands[1].npc) end
	end

	if targetPos and dist(targetPos) > 25 then teleportTo(targetPos) task.wait(0.4) end

	-- A. getQuests tại chỗ (chuẩn nhất)
	local raw = commF("getQuests")
	local function readQuests(data)
		local got = {}
		if type(data) ~= "table" then return got end
		local function read(e)
			if type(e) == "string" then
				local b, l = e:match("^(.-)%s*[%[Lv%.?%s]*(%d+)")
				got[#got + 1] = { id = e, mob = (b or e):gsub("%s+$", ""), lv = tonumber(l) or 0, num = 1 }
				return
			end
			if type(e) ~= "table" then return end
			local id, qlv, num, mob
			for k, v in pairs(e) do
				local ks = tostring(k):lower()
				if type(v) == "string" then
					if ks:find("quest") or ks == "name" or ks == "title" then id = id or v end
					if ks == "mob" or ks == "monster" or ks == "target" then mob = mob or v end
				elseif type(v) == "number" then
					if ks:find("level") or ks == "lv" or ks == "lvl" then qlv = qlv or v
					elseif ks:find("num") or ks:find("index") or ks == "id" then num = num or v end
				end
			end
			if id then got[#got + 1] = { id = id, mob = mob, lv = qlv or 0, num = num or 1 } end
		end
		local seqOk = pcall(function() for i = 1, #data do read(data[i]) end end)
		if not seqOk or #got == 0 then for _, e in pairs(data) do read(e) end end
		table.sort(got, function(a, b) return math.abs((a.lv or 0) - lv) < math.abs((b.lv or 0) - lv) end)
		return got
	end
	local list = readQuests(raw)
	local pick = list[1]
	if not pick then
		local cands = questCandidates(lv)
		if cands[1] then
			pick = { id = cands[1].id, num = cands[1].num or 1, mob = cands[1].mob }
		end
	end
	if pick and pick.id then
		commF("StartQuest", pick.id, pick.num or 1)
		task.wait(0.7)
		ok = questVisible()
		-- brute force: thử các id gần level nếu id đầu không ăn
		if not ok then
			for i = 2, math.min(6, #questCandidates(lv)) do
				local c = questCandidates(lv)[i]
				if c and c.id then
					commF("StartQuest", c.id, c.num or 1)
					task.wait(0.6)
					ok = questVisible()
					if ok then pick = { id = c.id, num = c.num or 1 } break end
				end
			end
		end
	end

	-- B. bấm E + nút hội thoại (cuối cùng)
	if not ok then
		pressKey(Enum.KeyCode.E, 0.12)
		task.wait(0.4)
		local bestBtn
		for _, b in ipairs(LPR.PlayerGui:GetDescendants()) do
			if b:IsA("TextButton") and b.Visible and b.AbsoluteSize.X > 20 then
				local pth = b:GetFullName()
				if not pth:find("SikeHub") then
					local t = (b.Text or ""):lower()
					if t:find("quest") or t:find("accept") or t:find("nhận") or t:find("%d") then bestBtn = b break end
				end
			end
		end
		if bestBtn then
			local p = bestBtn.AbsolutePosition + bestBtn.AbsoluteSize / 2
			safe(function()
				VIM:SendMouseButtonEvent(p.X, p.Y, 0, true, game, 0)
				task.wait(0.05)
				VIM:SendMouseButtonEvent(p.X, p.Y, 0, false, game, 0)
			end)
			task.wait(0.6)
			ok = questVisible()
		end
	end

	if ok then
		NOTIFY("Quest", "Đã nhận: " .. tostring(questTitle() or "?"), 4, Color3.fromRGB(90, 220, 130))
	else
		NOTIFY("Quest", "Chưa nhận được quest (thử lại hoặc dùng nút 'Nhận quest ngay')", 5, Color3.fromRGB(240, 160, 80))
	end
	return ok
end

-- Bọc pcall: lỗi giữa đường sẽ tự mở khoá cờ busy + báo lên HUD (không bao giờ kẹt)
local lastAcceptTry = tick() - 1000
local acceptFails = 0
local function acceptQuest()
	if not busyOK() then return false end
	-- chống spam: 8s, thất bại thì giãn dần tối đa 45s (an toàn với server)
	local gap = math.min(45, 8 * (acceptFails + 1))
	if tick() - lastAcceptTry < gap then return false end
	lastAcceptTry = tick()
	setBusy(true)
	local ok, res = pcall(acceptQuestBody)
	setBusy(false)
	if not ok then
		acceptFails = acceptFails + 1
		if acceptFails % 3 == 1 then
			NOTIFY("Quest", "Loi khi nhan quest: " .. tostring(res), 6, Color3.fromRGB(240, 120, 120))
		end
		return false
	end
	if res then acceptFails = 0 else acceptFails = acceptFails + 1 end
	return res
end
local function abandonQuest()
	commF("AbandonQuest")
	local b = guiNode("abandonBtn", 2)
	if b and b:IsA("TextButton") then
		local p = b.AbsolutePosition + b.AbsoluteSize / 2
		safe(function()
			VIM:SendMouseButtonEvent(p.X, p.Y, 0, true, game, 0)
			task.wait(0.05)
			VIM:SendMouseButtonEvent(p.X, p.Y, 0, false, game, 0)
		end)
	end
	NOTIFY("Quest", "Đã bỏ quest", 3)
end

local function pickEnemy()
	local q = questInfo()
	if CFG.FarmMode == "Theo tên mob" and CFG.FarmMobName ~= "" then
		return findMobByName(CFG.FarmMobName)
	end
	local lv = lvl()
	local want = q and q.mob or nil
	local best, score
	for _, e in ipairs(scanEnemies()) do
		local d = (e.hrp.Position - pos()).Magnitude
		if d < 2500 then
			local s = -d / 20
			if want and e.base:lower():find(want:lower(), 1, true) then s = s + 800 end
			if e.lv and e.lv > lv + 40 then s = s - 500 end
			if CFG.FarmMode == "Quest" and want and not e.base:lower():find(want:lower(), 1, true) then s = s - 300 end
			if not score or s > score then best, score = e, s end
		end
	end
	return best
end

local function chase(e)
	local mr = e.hrp
	local d = (mr.Position - pos()).Magnitude
	if d > CFG.AttackRange then
		local r = root()
		if r then
			safe(function()
				r.CFrame = CFrame.new(r.Position, Vector3.new(mr.Position.X, r.Position.Y, mr.Position.Z))
			end)
		end
		local hv = hum()
		if hv then hv.WalkSpeed = CFG.WalkSpeed end
		if d > CFG.AttackRange * 1.6 then teleportTo(mr.Position + Vector3.new(0, 4, -3)) end
	end
	if CFG.BringMob then safe(function() mr.CFrame = CFrame.new(pos() + Vector3.new(0, 0, -4)) end) end
	return (mr.Position - pos()).Magnitude
end

local lastBuso, lastKen = tick() - 1000, tick() - 1000
local function tickFarm()
	if not busyOK() then return end
	local hv = hum()
	if CFG.SafeMode and hv and hv.MaxHealth > 0 and hv.Health / hv.MaxHealth * 100 < CFG.SafeHP then return end

	if CFG.AutoSeaBeast then
		local e = nearestEnemy(function(x)
			local n = (x.name or ""):lower()
			return n:find("sea beast") or n:find("terror") or n:find("beast")
		end)
		if e then chase(e) return end
	end
	if CFG.AutoBoss then
		local e = nearestEnemy(function(x) return isBoss(x) end, 4000)
		if e then chase(e) return end
	end
	if CFG.AutoFarmLevel or CFG.AutoFarmNoQuest then
		if CFG.AutoFarmLevel and CFG.AutoQuest and not questVisible() and not questTitle() then
			acceptQuest()
			return
		end
		local e = pickEnemy()
		if e then chase(e) end
	end
end

local function tickKillAura()

	local e = nearestEnemy(nil, CFG.KillAuraRange)
	if e then chase(e) end
end

local function tickEquip()
	if not (CFG.AutoEquip or CFG.AutoMastery) then return end
	local c = LPR.Character
	if not c then return end
	local function owned()
		local t = {}
		for _, x in ipairs(LPR.Backpack:GetChildren()) do
			if x:IsA("Tool") and not x.Name:lower():find("fruit") then
				t[#t + 1] = { tool = x, lv = toolLevel(x) }
			end
		end
		return t
	end
	local eq = c:FindFirstChildOfClass("Tool")
	local want
	if CFG.AutoMastery and CFG.MasteryWeapon ~= "" then
		local list = owned()
		for _, x in ipairs(list) do
			if x.tool.Name == CFG.MasteryWeapon then want = x.tool break end
		end
		if not want then
			for _, x in ipairs(LPR.Backpack:GetChildren()) do
				if x:IsA("Tool") and x.Name == CFG.MasteryWeapon then want = x break end
			end
		end
	elseif CFG.AutoEquip then
		if eq then return end
		local list = owned()
		local mode = CFG.EquipMode
		if mode == "Best" then
			local best
			for _, x in ipairs(list) do if not best or x.lv > best.lv then best = x end end
			want = best and best.tool
		else
			for _, x in ipairs(list) do
				local n = x.tool.Name:lower()
				if mode == "Melee" and (n:find("combat") or n:find("black leg") or n:find("electro") or n:find("karate") or n:find("dragon talon") or n:find("superhuman")) then want = x.tool break end
				if mode == "Sword" and (n:find("sword") or n:find("blade") or n:find("katana") or n:find("saber") or n:find("cutlass") or n:find("scythe")) then want = x.tool break end
				if mode == "Gun" and (n:find("gun") or n:find("pistol") or n:find("rifle") or n:find("bazooka") or n:find("cannon") or n:find("slingshot")) then want = x.tool break end
				if mode == "Fruit" and n:find("fruit") then want = x.tool break end
			end
		end
	end
	if want and (not eq or eq ~= want) then safe(function() hum():EquipTool(want) end) end
end

local function tickChest()
	if not CFG.AutoChest or not busyOK() or tick() - lastChest < 1.2 then return end
	local best, bd
	for _, c in ipairs(chestsInWorld()) do
		local d = (c.pos - pos()).Magnitude
		if d < 900 and (not bd or d < bd) then best, bd = c, d end
	end
	if not best then return end
	lastChest = tick()
	local r = root()
	if CFG.ChestMode == "Teleport" then
		teleportTo(best.pos)
	else
		local tgt = CFrame.new(best.pos + Vector3.new(0, 4, 0))
		for i = 1, 12 do
			safe(function() r.CFrame = r.CFrame:Lerp(tgt, 0.35) end)
			task.wait(0.06)
		end
	end
	task.wait(0.3)
	safe(function()
		local cd = best.inst:FindFirstChildWhichIsA("ClickDetector", true)
		if cd and API.firecd then fireclickdetector(cd) end
		local pp = best.inst:FindFirstChildWhichIsA("ProximityPrompt", true)
		if pp and API.firepp then fireproximityprompt(pp) end
	end)
end
local function tickFruit()
	if not CFG.AutoFruit and not CFG.FruitSniper then return end
	if not busyOK() or tick() - lastFruit < 1.5 then return end
	local list = fruitsInWorld()
	local best, bd
	for _, f in ipairs(list) do
		local d = (f.pos - pos()).Magnitude
		if d < 5000 then
			if CFG.FruitSniper then
				if fruitRarity(f.name) >= CFG.FruitSniperRarity and (not bd or d < bd) then best, bd = f, d end
			elseif not bd or d < bd then best, bd = f, d end
		end
	end
	if not best then return end
	lastFruit = tick()
	NOTIFY("Fruit", best.name .. " (" .. math.floor(bd) .. "m)", 4, Color3.fromRGB(255, 120, 240))
	teleportTo(best.pos)
end
local function tickFruitBring()
	if not CFG.FruitBring then return end
	local p = pos()
	for _, f in ipairs(fruitsInWorld()) do

local function tickStoreFruit()
	if not CFG.AutoStoreFruit or tick() - lastStore < 5 then return end
	local inv = commF("GetFruits")
	if type(inv) ~= "table" then return end
	for _, f in ipairs(inv) do
		local nm = type(f) == "table" and (f.Name or f.name) or nil
		local ra = type(f) == "table" and tonumber(f.Rarity or f.rarity) or 0
		if nm and (nm == CFG.StoreFruitPick or (ra or 0) >= CFG.StoreRarity) then
			lastStore = tick()
			commF("StoreFruit", nm)
			NOTIFY("Kho", "Cất fruit: " .. nm, 4, Color3.fromRGB(140, 200, 255))
			return
		end
	end
end
local dropNotified = {}
local function tickDrop()
	if not CFG.AutoDrop or not busyOK() or tick() - lastDrop < 1.5 then return end
	local r = root()
	if not r then return end
	for _, kw in ipairs(BF_OFFSETS.items.dropKeywords or {}) do
		local k = kw:lower()
		local found
		local function look(inst, depth)
			if found or depth > 2 then return end
			if (inst:IsA("Tool") or inst:IsA("BasePart")) and inst.Name:lower():find(k, 1, true)
				and not inst:IsDescendantOf(LPR.Character) and inst.Name:lower() ~= k then
				local p = partPos(inst)
				if p and (p - r.Position).Magnitude < 900 then found = { inst = inst, pos = p } return end
			end
			for _, c in ipairs(inst:GetChildren()) do look(c, depth + 1) end
		end
		for _, c in ipairs(workspace:GetChildren()) do look(c, 0) end
		if found then
			lastDrop = tick()
			if not dropNotified[found.inst] then
				dropNotified[found.inst] = true
				NOTIFY("Đồ rơi", found.inst.Name, 4, Color3.fromRGB(255, 210, 120))
			end
			teleportTo(found.pos)
			return
		end
	end
end
local function tickPart()
	if not CFG.AutoPart or not busyOK() or tick() - lastPart < 2 then return end
	local sea = onSea()
	local list = BF_OFFSETS.fruitParts[sea] or {}
	if #list == 0 then return end
	local best, bd
	for _, t in ipairs(list) do
		local p = V(t)
		if p then
			local d = (p - pos()).Magnitude
			if d < 400 then
				lastPart = tick()
				teleportTo(p)
				NOTIFY("Mảnh Fruit", "Nhặt mảnh gần đây!", 3)
				return
			end
			if d > 100 and (not bd or d < bd) then best, bd = p, d end
		end
	end
	if best and bd < 40000 then lastPart = tick() teleportTo(best) end
end
local function tickChip()
	if not CFG.AutoChip or tick() - (CFG._chipAt or 0) < 3 then return end
	CFG._chipAt = tick()
	local r = root()
	if not r then return end
	local found
	for _, c in ipairs(workspace:GetChildren()) do
		if c:IsA("Tool") or c:IsA("Model") or c:IsA("BasePart") then
			local n = c.Name:lower()
			if n:find("chip") or n:find("microchip") or n:find("fragment of") then
				local p = partPos(c)
				if p and (p - r.Position).Magnitude < 3000 then found = { inst = c, pos = p } break end
			end
		end
	end
	if found then
		NOTIFY("Raid", "Nhặt chip: " .. found.inst.Name, 4, Color3.fromRGB(255, 160, 90))
		teleportTo(found.pos)
	end
end

local espCache = {}
local function espClear()
	for inst, e in pairs(espCache) do
		safe(function() e.bb:Destroy() end)
		safe(function() if e.hl then e.hl:Destroy() end end)
		safe(function() if e.dr then e.dr:Destroy() end end)

	end

end
local function espTrack(inst, text, color)
	if not inst or not inst.Parent then return end
	local e = espCache[inst]
	if not e then
		local bb = Instance.new("BillboardGui")
		bb.Name = "SikeESP"
		bb.Size = UDim2.new(0, 180, 0, 30)
		bb.StudsOffset = Vector3.new(0, 3, 0)
		bb.AlwaysOnTop = true
		bb.MaxDistance = CFG.ESPRange
		local lbl = Instance.new("TextLabel")
		lbl.Size = UDim2.new(1, 0, 1, 0)
		lbl.BackgroundTransparency = 0.35
		lbl.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
		lbl.TextColor3 = color
		lbl.Font = Enum.Font.GothamBold
		lbl.TextSize = 12
		lbl.TextStrokeTransparency = 0.5
		lbl.Parent = bb
		local hl = Instance.new("Highlight")
		hl.FillColor = color
		hl.FillTransparency = 0.78
		hl.OutlineColor = Color3.fromRGB(255, 255, 255)
		hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
		hl.Parent = inst
		bb.Parent = inst

		espCache[inst] = e
	end
	e.label.Text = text
	e.label.TextColor3 = color
	e.label.Parent.MaxDistance = CFG.ESPRange
	return e
end
-- kiểm tra instance còn trong DataModel không (chắc hơn IsDescendantOf ở vài executor)
local function espAlive(inst)
	if not inst or not inst.Parent then return false end
	local ok, res = pcall(function() return inst:IsDescendantOf(game) end)
	if ok and res then return true end
	local cur, n = inst.Parent, 0
	while cur and n < 16 do cur = cur.Parent n = n + 1 end
	return cur == nil
end
local function espUpdate()
	local anyOn = CFG.ESPPlayers or CFG.ESPMobs or CFG.ESPBoss or CFG.ESPChests or CFG.ESPFruits or CFG.ESPNpcs
	if not anyOn then if next(espCache) then espClear() end return end
	for inst, e in pairs(espCache) do
		if not espAlive(inst) then
			safe(function() e.bb:Destroy() end)
			safe(function() if e.hl then e.hl:Destroy() end end)
			espCache[inst] = nil
		end
	end
	if CFG.ESPPlayers then
		for _, pl in ipairs(Players:GetPlayers()) do
			if pl ~= LPR and pl.Character then
				local h = pl.Character:FindFirstChildOfClass("Humanoid")
				local r = pl.Character:FindFirstChild("HumanoidRootPart")
				if h and r and h.Health > 0 then
					local l = plLevel(pl)
					espTrack(r, string.format("%s [Lv.%d] %d%% · %dm", pl.Name, l,
						math.floor(h.Health / math.max(1, h.MaxHealth) * 100), dist(r.Position)),
						Color3.fromRGB(90, 255, 130))
				end
			end
		end
	end
	if CFG.ESPMobs or CFG.ESPBoss then
		local n = 0
		for _, e in ipairs(scanEnemies()) do
			n = n + 1
			if n > 40 then break end
			local d = (e.hrp.Position - pos()).Magnitude
			if d < CFG.ESPRange then
				local boss = isBoss(e)
				if (boss and CFG.ESPBoss) or ((not boss) and CFG.ESPMobs) then
					espTrack(e.hrp, string.format("%s%s · %dm", e.name, boss and " [BOSS]" or "", d),
						boss and Color3.fromRGB(255, 80, 80) or Color3.fromRGB(255, 175, 70))
				end
			end
		end
	end
	if CFG.ESPChests then
		for _, c in ipairs(chestsInWorld()) do
			local d = (c.pos - pos()).Magnitude
			if d < CFG.ESPRange then
				espTrack(c.inst, string.format("RƯƠNG · %dm", math.floor(d)), Color3.fromRGB(255, 230, 90))
			end
		end
	end
	if CFG.ESPFruits then
		for _, f in ipairs(fruitsInWorld()) do
			local d = (f.pos - pos()).Magnitude
			if d < CFG.ESPRange then
				espTrack(f.inst, string.format("%s (r%d) · %dm", f.name, fruitRarity(f.name), math.floor(d)),
					Color3.fromRGB(255, 110, 240))
			end
		end
	end
	if CFG.ESPNpcs then
		for _, fname in ipairs({ "NPCs" }) do
			local f = workspace:FindFirstChild(fname)
			if f then
				for _, m in ipairs(f:GetChildren()) do
					local hrp = m:FindFirstChild("HumanoidRootPart")
					if hrp then
						local d = (hrp.Position - pos()).Magnitude
						if d < CFG.ESPRange then
							espTrack(hrp, string.format("%s · %dm", m.Name, math.floor(d)),
								Color3.fromRGB(120, 200, 255))
						end
					end
				end
			end
		end
	end
end

local flyBV, flyBG
local function flySet(on)
	if on then
		local r = root()
		if not r then CFG.Fly = false return end
		if flyBV and flyBV.Parent ~= r then flySet(false) end
		if not flyBV then
			flyBV = Instance.new("BodyVelocity")
			flyBV.MaxForce = Vector3.new(9e9, 9e9, 9e9)
			flyBV.Velocity = Vector3.new(0, 0, 0)
			flyBV.P = 1e4
			flyBV.Parent = r

			flyBG.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
			flyBG.P = 9e4
			flyBG.Parent = r

		if flyBV then safe(function() flyBV:Destroy() end) flyBV = nil end
		if flyBG then safe(function() flyBG:Destroy() end) flyBG = nil end
	end
end
local function flyStep()
	if not CFG.Fly then return end
	local r = root()
	if not r then return end
	if not flyBV or flyBV.Parent ~= r then flySet(true) return end
	local cam = workspace.CurrentCamera
	local fwd, right = cam.CFrame.LookVector, cam.CFrame.RightVector
	local dir = Vector3.new(0, 0, 0)
	local key = false
	if UIS.IsKeyDown and safe(function() return UIS:IsKeyDown(Enum.KeyCode.W) end) then dir = dir + fwd key = true end
	if UIS.IsKeyDown and safe(function() return UIS:IsKeyDown(Enum.KeyCode.S) end) then dir = dir - fwd key = true end
	if UIS.IsKeyDown and safe(function() return UIS:IsKeyDown(Enum.KeyCode.A) end) then dir = dir - right key = true end
	if UIS.IsKeyDown and safe(function() return UIS:IsKeyDown(Enum.KeyCode.D) end) then dir = dir + right key = true end
	if safe(function() return UIS:IsKeyDown(Enum.KeyCode.Space) end) then dir = dir + Vector3.new(0, 1, 0) key = true end
	if safe(function() return UIS:IsKeyDown(Enum.KeyCode.LeftShift) end) then dir = dir - Vector3.new(0, 1, 0) key = true end
	if not key then dir = fwd end
	if dir.Magnitude > 0 then dir = dir.Unit end
	safe(function()
		flyBV.Velocity = dir * CFG.FlySpeed
		flyBG.CFrame = CFrame.lookAt(r.Position, r.Position + dir)
	end)
end

local homePlace = game.PlaceId
local function serverHop()
	if not API.request then NOTIFY("Server", "Executor thiếu API request → dùng Rejoin / Join JobId", 6, Color3.fromRGB(255, 120, 120)) return end
	NOTIFY("Server", "Đang dò server...", 4)
	local done = false
	safe(function()
		local url = (BF_OFFSETS.server.publicServersApi:gsub("{PlaceId}", tostring(homePlace)))
		local res = (typeof(request) == "function") and request({ Url = url, Method = "GET" })
			or http_request({ Url = url, Method = "GET" })
		local data = HTS:JSONDecode(res.Body)
		local rows = data.data or {}
		table.sort(rows, function(a, b) return (a.playing or 0) < (b.playing or 0) end)
		for _, s in ipairs(rows) do
			local cnt, maxp = s.playing or 0, s.maxPlayers or 12
			if s.id and s.id ~= game.JobId and cnt < maxp - 1 and cnt >= CFG.HopMinPlayers then
				done = true
				NOTIFY("Server", string.format("Vào server %d/%d", cnt, maxp), 4)
				TPS:TeleportToPlaceInstance(homePlace, s.id, LPR)
				return
			end
		end
	end)
	if not done then
		NOTIFY("Server", "Chưa có server trống → thử lại sau 8s", 5, Color3.fromRGB(255, 190, 90))
		task.delay(8, function() task.spawn(serverHop) end)
	end
end
local function rejoin() NOTIFY("Server", "Rejoin...", 3) TPS:Teleport(homePlace, LPR) end
local function joinCode(code)
	code = (code or ""):gsub("%s", "")
	if code == "" then NOTIFY("Server", "Dán JobId vào ô trước", 4) return end
	NOTIFY("Server", "Join: " .. code, 4)
	TPS:TeleportToPlaceInstance(homePlace, code, LPR)
end
local function autoRejoinStep()
	if not CFG.AutoRejoin or tick() - (CFG._rejoinAt or 0) < 15 then return end
	local kicked = false
	if safe(function()
		for _, d in ipairs(game:GetService("CoreGui"):GetDescendants()) do
			if d.Name == "ErrorFrame" or d.Name == "DisconnectFrame" then return true end
		end
		return false
	end) then kicked = true end
	if safe(function()
		local e = LPR.PlayerGui.Main:FindFirstChild("Error")
		return e ~= nil and e.Visible == true
	end) then kicked = true end
	if kicked then
		CFG._rejoinAt = tick()
		NOTIFY("Server", "Bị kick → auto rejoin sau 5s", 6, Color3.fromRGB(255, 140, 140))
		task.delay(5, function() safe(rejoin) end)
	end
end

local function mkSlider(sec, label, key, min, max, cb, isInt)
	local f = elFrame(sec, 34)
	local lbl = Instance.new("TextLabel")
	lbl.Size = UDim2.new(1, -70, 0, 14)
	lbl.Position = UDim2.new(0, 8, 0, 1)
	lbl.BackgroundTransparency = 1
	lbl.Text = label
	lbl.TextColor3 = Color3.fromRGB(226, 226, 236)
	lbl.TextXAlignment = Enum.TextXAlignment.Left
	lbl.Font = Enum.Font.Gotham
	lbl.TextSize = 11 * UISCALE
	lbl.Parent = f
	local val = Instance.new("TextBox")
	val.Size = UDim2.new(0, 54 * UISCALE, 0, 15 * UISCALE)
	val.Position = UDim2.new(1, -58 * UISCALE, 0, 1)
	val.BackgroundColor3 = Color3.fromRGB(22, 20, 34)
	val.Text = tostring(CFG[key])
	val.TextColor3 = TH.accent
	val.Font = Enum.Font.Code
	val.TextSize = 11 * UISCALE
	val.ClearTextOnFocus = false
	val.Parent = f
	addCorner(val, 4)
	local bar = Instance.new("TextButton")
	bar.Size = UDim2.new(1, -16, 0, 10 * UISCALE)
	bar.Position = UDim2.new(0, 8, 0, 20 * UISCALE)
	bar.BackgroundColor3 = Color3.fromRGB(48, 46, 66)
	bar.Text = ""
	bar.AutoButtonColor = false
	bar.Parent = f
	addCorner(bar, 5)
	local fill = Instance.new("Frame")
	fill.Size = UDim2.new(0, 0, 1, 0)
	fill.BackgroundColor3 = TH.accent
	fill.BorderSizePixel = 0
	fill.Parent = bar
	addCorner(fill, 5)
	local function setFromX(px)
		local rel = clamp((px - bar.AbsolutePosition.X) / math.max(1, bar.AbsoluteSize.X), 0, 1)
		local v = min + (max - min) * rel
		if isInt then v = math.floor(v + 0.5) end
		CFG[key] = v
		val.Text = tostring(v)
		fill.Size = UDim2.new(rel, 0, 1, 0)
		if cb then safe(cb, v) end
	end
	local dragging = false
	bar.InputBegan:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
			dragging = true setFromX(i.Position.X)
		end
	end)
	bar.InputChanged:Connect(function(i)
		if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
			setFromX(i.Position.X)
		end
	end)
	bar.InputEnded:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then dragging = false end
	end)
	val.FocusLost:Connect(function()
		local n = tonumber(val.Text)
		if n then
			n = clamp(n, min, max)
			CFG[key] = isInt and math.floor(n) or n
			val.Text = tostring(CFG[key])
			local rel = (CFG[key] - min) / math.max(1e-6, (max - min))
			fill.Size = UDim2.new(clamp(rel, 0, 1), 0, 1, 0)
			if cb then safe(cb, CFG[key]) end
		else
			val.Text = tostring(CFG[key])
		end
	end)
	local initRel = (CFG[key] - min) / math.max(1e-6, (max - min))
	fill.Size = UDim2.new(clamp(initRel, 0, 1), 0, 1, 0)
	return f
end
